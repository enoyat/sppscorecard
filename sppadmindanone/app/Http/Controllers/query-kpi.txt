create view vunittype as (
SELECT idforklifttype, forklifttype.namaforklifttype, count(kdunit) as jmlunit, idcbu,idregion,idsitename from unit join forklifttype on unit.idforklifttype=forklifttype.id group by idforklifttype);
create view vpa as (select idforklifttype, periode, sum(planunitkerja) as sumplanunitkerja, sum(totaljamkerja) as sumtotaljamkerja, avg(paforklift) as avgpaforklift, physicalavailable.idcbu,physicalavailable.idregion, physicalavailable.idsitename
from physicalavailable join unit on unit.kdunit=physicalavailable.kdunit
group by periode,idforklifttype);
